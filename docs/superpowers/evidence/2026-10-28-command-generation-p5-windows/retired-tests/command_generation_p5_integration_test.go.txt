var updateP5Policy = flag.Bool("update-p5-policy", false, "record the P5 evaluation policy manifest after reviewing source changes")

// P5 compares the current product with command-simplification off (B) against
// the same product with once (C). The historical A/A1 pair is run from the
// pre-P2 worktree so that it keeps a fixed analyzer and authorization version.
// A policy manifest makes later source edits invalidate the saved comparison.
const p5PolicyVersion = 1

type p5PolicyManifest struct {
	Version      int               `json:"version"`
	SourceHashes map[string]string `json:"source_hashes"`
	PromptHashes map[string]string `json:"prompt_hashes"`
}

func p5PolicyPaths(t testing.TB) []string {
	t.Helper()
	root := filepath.Join("..", "..")
	var paths []string
	for _, dir := range []string{"internal/approval", "internal/commandanalysis", "internal/prompts"} {
		err := filepath.WalkDir(filepath.Join(root, dir), func(path string, entry fs.DirEntry, err error) error {
			if err != nil {
				return err
			}
			if entry.IsDir() {
				return nil
			}
			name := entry.Name()
			if strings.HasSuffix(name, "_test.go") {
				return nil
			}
			switch filepath.Ext(name) {
			case ".go", ".ps1", ".md":
			default:
				return nil
			}
			rel, err := filepath.Rel(root, path)
			if err != nil {
				return err
			}
			paths = append(paths, filepath.ToSlash(rel))
			return nil
		})
		require.NoError(t, err)
	}
	paths = append(paths,
		"internal/app/command_analysis.go",
		"internal/app/command_organization.go",
		"internal/app/command_payload.go",
		"internal/app/command_reorganization.go",
		"internal/app/completion.go",
		"internal/app/completion_checkpoint.go",
		"internal/app/mods.go",
		"internal/app/request_session.go",
		"internal/app/review.go",
		"internal/app/review_pages.go",
		"internal/app/review_presentation.go",
		"internal/app/review_summary.go",
		"internal/app/stream_runner.go",
		"internal/app/tool_selection.go",
		"internal/app/tool_support.go",
		"internal/config/config.go",
		"internal/config/config_template.yml",
		"internal/config/save.go",
		"internal/evals/command_generation.go",
		"internal/evals/execution.go",
		"internal/tools/guidance.go",
		"internal/tools/shell.go",
	)
	sort.Strings(paths)
	unique := paths[:0]
	for _, path := range paths {
		if len(unique) == 0 || unique[len(unique)-1] != path {
			unique = append(unique, path)
		}
	}
	return unique
}

func buildP5PolicyManifest(t testing.TB) p5PolicyManifest {
	t.Helper()
	root := filepath.Join("..", "..")
	manifest := p5PolicyManifest{
		Version:      p5PolicyVersion,
		SourceHashes: make(map[string]string),
		PromptHashes: make(map[string]string),
	}
	for _, rel := range p5PolicyPaths(t) {
		data, err := os.ReadFile(filepath.Join(root, filepath.FromSlash(rel)))
		require.NoError(t, err)
		digest := sha256.Sum256(data)
		manifest.SourceHashes[rel] = hex.EncodeToString(digest[:])
	}
	for name, text := range map[string]string{
		"minimal":                prompts.Minimal,
		"powershell_intent":      prompts.PowerShellIntentGuidance,
		"query_policy":           prompts.QueryPolicy,
		"script_policy":          prompts.ScriptPolicy,
		"tool_selection_general": prompts.ToolSelectionGeneral,
	} {
		digest := sha256.Sum256([]byte(text))
		manifest.PromptHashes[name] = hex.EncodeToString(digest[:])
	}
	return manifest
}

func readP5PolicyManifest(t testing.TB) p5PolicyManifest {
	t.Helper()
	data, err := os.ReadFile(filepath.Join("testdata", "command_generation", "p5_policy.json"))
	require.NoError(t, err)
	var manifest p5PolicyManifest
	require.NoError(t, json.Unmarshal(data, &manifest))
	require.NotEmpty(t, manifest.SourceHashes)
	require.NotEmpty(t, manifest.PromptHashes)
	return manifest
}

func requireP5PolicyMatches(t testing.TB, want, got p5PolicyManifest) {
	t.Helper()
	require.Equal(t, want.Version, got.Version)
	compare := func(kind string, want, got map[string]string) {
		t.Helper()
		var missing, extra, changed []string
		for path, digest := range want {
			actual, ok := got[path]
			if !ok {
				missing = append(missing, path)
				continue
			}
			if actual != digest {
				changed = append(changed, path)
			}
		}
		for path := range got {
			if _, ok := want[path]; !ok {
				extra = append(extra, path)
			}
		}
		sort.Strings(missing)
		sort.Strings(extra)
		sort.Strings(changed)
		require.Empty(t, missing, "%s missing from current policy: %s", kind, strings.Join(missing, ", "))
		require.Empty(t, extra, "%s added since the recorded policy: %s", kind, strings.Join(extra, ", "))
		require.Empty(t, changed, "%s changed since the recorded policy; review and rerun with -update-p5-policy: %s", kind, strings.Join(changed, ", "))
	}
	compare("source", want.SourceHashes, got.SourceHashes)
	compare("prompt", want.PromptHashes, got.PromptHashes)
}

// TestCommandGenerationP5PolicyManifest is a no-key guard for the files whose
// behavior the real-model comparison depends on.
func TestCommandGenerationP5PolicyManifest(t *testing.T) {
	got := buildP5PolicyManifest(t)
	path := filepath.Join("testdata", "command_generation", "p5_policy.json")
	if *updateP5Policy {
		data, err := json.MarshalIndent(got, "", "  ")
		require.NoError(t, err)
		require.NoError(t, os.WriteFile(path, append(data, '\n'), 0600))
		t.Logf("recorded %s with %d source hashes", path, len(got.SourceHashes))
		return
	}
	requireP5PolicyMatches(t, readP5PolicyManifest(t), got)
}

func p5PolicyFingerprint(t testing.TB) string {
	t.Helper()
	got := buildP5PolicyManifest(t)
	requireP5PolicyMatches(t, readP5PolicyManifest(t), got)
	data, err := json.Marshal(got)
	require.NoError(t, err)
	digest := sha256.Sum256(data)
	return hex.EncodeToString(digest[:])
}

type p5GenerationArm struct {
	Name           string
	Simplification config.CommandSimplification
}

func p5GenerationArms() []p5GenerationArm {
	return []p5GenerationArm{
		{Name: "B", Simplification: config.CommandSimplificationOff},

	}
}

type p5GenerationReport struct {
	GeneratedAt       string          `json:"generated_at"`
	Commit            string          `json:"commit,omitempty"`
	GoVersion         string          `json:"go_version"`
	OS                string          `json:"os"`
	Model             string          `json:"model"`
	Phase             string          `json:"phase"`
	Minimal           bool            `json:"minimal"`
	Repeats           int             `json:"repeats"`
	PolicyFingerprint string          `json:"policy_fingerprint"`
	Runs              []generationRun `json:"runs"`
}

type p5PreparedRun struct {
	m            *Mods
	cfg          *config.Config
	registry     *toolregistry.Registry
	client       stream.Client
	messages     []proto.Message
	specs        []proto.ToolSpec
	organization *commandOrganizationState
}

func prepareP5Run(t *testing.T, c evals.CommandGenerationCase, root string, arm p5GenerationArm, minimal bool, model string, ctx context.Context) *p5PreparedRun {
	t.Helper()
	m := testIntegrationModsWithBaseURL(t, "deepseek", model, "https://api.deepseek.com/")
	cfg := defaultConfig()
	cfg.APIs = m.Config.APIs
	cfg.API = "deepseek"
	cfg.Model = model
	cfg.WorkingDir = root
	cfg.NoSave = true
	cfg.NoInstructions = true
	cfg.Raw = true
	cfg.Minimal = minimal
	cfg.Format = ""
	cfg.Role = ""
	cfg.Think = false
	cfg.ReviewMode = ReviewAuto
	cfg.InteractiveTTYAvailable = false
	cfg.CommandSimplification = arm.Simplification
	cfg.BuiltinTools.Filesystem = FilesystemAlways
	cfg.BuiltinTools.Shell = true
	cfg.BuiltinTools.ShellReadOnlyCommands = nil
	cfg.SkillsDirs = []string{filepath.Join(root, "absent-skills")}
	cfg.MCPServers = nil
	cfg.WebSearch = false
	m.Config = &cfg
	m.reviewer = newToolReviewer(&cfg)
	m.ctx = ctx

	require.NoError(t, m.setupStreamContext(c.Prompt))
	api, mod, err := m.resolveModel(&cfg)
	require.NoError(t, err)
	configs, err := m.buildProviderConfigs(mod, api)
	require.NoError(t, err)
	_, err = m.resolveThinkWithOllama(&mod, &configs.Anthropic, &configs.Google, &configs.Ollama, &configs.OpenAI)
	require.NoError(t, err)
	client, err := newStreamClient(modelProtocol(mod), configs.Anthropic, configs.Google, configs.Ollama, configs.OpenAI)
	require.NoError(t, err)
	registry, err := m.buildToolRegistryForProvider(ctx, &cfg, websearch.Config{}, c.Prompt, client)
	require.NoError(t, err)
	t.Cleanup(func() { require.NoError(t, registry.Close()) })
	require.NoError(t, m.injectToolSelectionPrompt(registry))

	var organization *commandOrganizationState
	if arm.Simplification == config.CommandSimplificationOnce {
		organization = newCommandOrganizationState(ctx, c.Prompt)
		m.replaceCommandOrganization(organization)
		t.Cleanup(m.closeCommandOrganization)
		// Production binds source context after the completion context is set up.
		organization.bindSourceContext(m.messages)
	}
	return &p5PreparedRun{
		m:            m,
		cfg:          &cfg,
		registry:     registry,
		client:       client,
		messages:     cloneCompletionMessages(m.messages),
		specs:        append([]proto.ToolSpec(nil), registry.Specs()...),
		organization: organization,
	}
}

// TestCommandGenerationP5AutomaticOrganizationWiring guards the C arm's
// connection to the production caller. It uses a fixed nested command and a
// fake registry, so it cannot execute model output or grant authorization.
func TestCommandGenerationP5AutomaticOrganizationWiring(t *testing.T) {
	cfg := defaultConfig()
	cfg.WorkingDir = t.TempDir()
	cfg.CommandSimplification = config.CommandSimplificationOnce
	cfg.ReviewMode = ReviewAuto
	cfg.InteractiveTTYAvailable = false
	m := &Mods{Config: &cfg, ctx: context.Background(), reviewer: newToolReviewer(&cfg)}
	state := newCommandOrganizationState(m.ctx, "Summarize project")
	defer state.close()
	var calls atomic.Int32
	registry := organizationRegistry(t, "shell_run", toolregistry.ToolKindShell, &calls)
	caller := m.toolResultCallerForOrganization(registry, &cfg, state)
	data, err := json.Marshal(map[string]string{"command": organizationCommand()})
	require.NoError(t, err)
	_, err = caller(proto.ToolCallRequest{ID: "p5-wiring", Name: "shell_run", Arguments: data})
	var control *commandReorganizationError
	require.ErrorAs(t, err, &control)
	require.True(t, control.Automatic)
	require.Zero(t, calls.Load())
}

func runP5GenerationComparison(t *testing.T, firstOnly bool) {
	skipIfNoKey(t, "DEEPSEEK_API_KEY", "P5 command generation")
	fingerprint := p5PolicyFingerprint(t)
	cases, err := evals.CommandGenerationCases()
	require.NoError(t, err)
	repeats := 5
	if value := os.Getenv("MODS_GENERATION_REPEATS"); value != "" {
		repeats, err = strconv.Atoi(value)
		require.NoError(t, err)
		require.True(t, repeats >= 1 && repeats <= 10)
	}
	minimal := os.Getenv("MODS_GENERATION_MINIMAL") == "1"
	model := os.Getenv("MODS_BLACKBOX_MODEL")
	if model == "" {
		model = "deepseek-v4-flash"
	}
	phase := "completion"
	if firstOnly {
		phase = "first-round"
	}
	var reports []generationRun
	t.Cleanup(func() {
		commandanalysis.CloseBridge()
		dir := os.Getenv("MODS_GENERATION_REPORT_DIR")
		if dir == "" {
			return
		}
		require.NoError(t, os.MkdirAll(dir, 0700))
		report := p5GenerationReport{
			GeneratedAt:       time.Now().UTC().Format(time.RFC3339),
			Commit:            os.Getenv("MODS_GENERATION_COMMIT"),
			GoVersion:         runtime.Version(),
			OS:                runtime.GOOS + "/" + runtime.GOARCH,
			Model:             model,
			Phase:             phase,
			Minimal:           minimal,
			Repeats:           repeats,
			PolicyFingerprint: fingerprint,
			Runs:              reports,
		}
		data, err := json.MarshalIndent(report, "", "  ")
		require.NoError(t, err)
		name := fmt.Sprintf("p5-%s-minimal-%t.json", phase, minimal)
		require.NoError(t, os.WriteFile(filepath.Join(dir, name), append(data, '\n'), 0600))
	})
	for _, c := range cases {
		// Completion only has an automatic numeric oracle for the explicit
		// tasks. The ambiguous original is retained for manual review; its
		// correctness remains unavailable rather than guessed.
		if !firstOnly && len(c.Expected) == 0 && c.ID != "ambiguous-code" {
			continue
		}
		if selected := os.Getenv("MODS_GENERATION_CASES"); selected != "" && !strings.Contains(","+selected+",", ","+c.ID+",") {
			continue
		}
		for repeat := 0; repeat < repeats; repeat++ {
			arms := p5GenerationArms()
			if repeat%2 == 1 {
				arms = []p5GenerationArm{arms[1], arms[0]}
			}
			root := t.TempDir()
			for _, arm := range arms {
				t.Run(fmt.Sprintf("%s/%d/%s", c.ID, repeat+1, arm.Name), func(t *testing.T) {
					// root is owned by this t.TempDir, never a user-provided path.
					require.NoError(t, os.RemoveAll(root))
					require.NoError(t, os.MkdirAll(root, 0700))
					for name, content := range evals.CommandGenerationFiles() {
						path := filepath.Join(root, filepath.FromSlash(name))
						require.NoError(t, os.MkdirAll(filepath.Dir(path), 0700))
						require.NoError(t, os.WriteFile(path, []byte(content), 0600))
					}
					ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
					defer cancel()
					prepared := prepareP5Run(t, c, root, arm, minimal, model, ctx)
					r := generationRun{
						Case:                  c.ID,
						Arm:                   arm.Name,
						Repeat:                repeat + 1,
						Model:                 model,
						OS:                    runtime.GOOS,
						Minimal:               minimal,
						CommandSimplification: string(arm.Simplification),
						ReviewMode:            "auto",
						PolicyFingerprint:     fingerprint,
						FirstRoundOnly:        firstOnly,
						Outcome:               evals.GenerationOutcome{Query: c.Query},
					}
					r.BaselineRevision = os.Getenv("MODS_GENERATION_BASELINE")
					if r.BaselineRevision == "" {
						r.BaselineRevision = "p5"
					}
					promptBytes, err := json.Marshal(struct {
						Messages []proto.Message
						Tools    []proto.ToolSpec
					}{prepared.messages, prepared.specs})
					require.NoError(t, err)
					r.PromptFingerprint = fmt.Sprintf("%x", sha256.Sum256(promptBytes))
					r.MaxTokens = 4096
					started := time.Now()
					defer func() {
						r.ElapsedMS = time.Since(started).Milliseconds()
						r.ZeroReviewCompletion = r.Outcome.ZeroReviewCompletion()
						reports = append(reports, r)
						data, _ := json.Marshal(r)
						t.Logf("P5 generation report: %s", data)
					}()
					maxTokens := int64(4096)
					req := proto.Request{Messages: prepared.messages, API: "deepseek", Model: model, Tools: prepared.specs, Temperature: ptrOrNil(float64(0)), MaxTokens: &maxTokens, TrackUsage: true}
					if !firstOnly {
						zero := 0
						r.Outcome.ActualReviewPrompts = &zero // noninteractive harness cannot display a review
						productionCaller := prepared.m.toolResultCallerForOrganization(prepared.registry, prepared.cfg, prepared.organization)
						req.ResultCaller = func(call proto.ToolCallRequest) (proto.ToolResult, error) {
							o := generationObserve(prepared.m, prepared.registry, call)
							r.Calls = append(r.Calls, o)
							r.Outcome.ToolCalls++
							if o.Effect == "write" {
								r.Outcome.UnexpectedWrites++
							}
							result, err := productionCaller(call)
							r.Outcome.ObserveReorganization(err)
							switch {
							case errors.Is(err, errReviewUnavailable):
								r.Outcome.ApprovalUnavailable++
							case err != nil:
								if _, control := evals.ReorganizationKind(err); !control && ctx.Err() == nil {
									r.Outcome.ToolErrors++
									if o.Decision == approval.DecisionAllow {
										r.Outcome.AllowPathErrors++
									}
								}
							}
							if o.Effect == "write" && err == nil {
								t.Errorf("write call executed during P5 completion: %s", call.Name)
							}
							return result, err
						}
					}
					st := prepared.client.Request(ctx, req)
					defer st.Close()
					for round := 0; round < 6; round++ {
						var answer strings.Builder
						for st.Next() {
							chunk, err := st.Current()
							if err != nil && !errors.Is(err, stream.ErrNoContent) {
								r.Error = err.Error()
								return
							}
							answer.WriteString(chunk.Content)
						}
						r.Answer = answer.String()
						usage := st.Usage()
						r.InputTokens = int64(usage.InputTokens)
						r.OutputTokens = int64(usage.OutputTokens)
						if err := st.Err(); err != nil {
							r.Error = err.Error()
							return
						}
						if firstOnly {
							for _, msg := range st.Messages() {
								if msg.Role != proto.RoleAssistant {
									continue
								}
								for _, call := range msg.ToolCalls {
									r.Calls = append(r.Calls, generationObserve(prepared.m, prepared.registry, proto.ToolCallRequest{Name: call.Function.Name, Arguments: call.Function.Arguments}))
								}
							}
							r.Outcome.ToolCalls = len(r.Calls)
							if len(r.Calls) > 0 {
								r.Outcome.ToolRounds = 1
							}
							return
						}
						results := st.CallTools()
						if len(results) == 0 {
							r.Outcome.Completed = true
							break
						}
						r.Outcome.ToolRounds++
					}
					if !r.Outcome.Completed {
						r.Error = "tool round limit reached"
					}
					correct := r.Outcome.Completed && evals.CheckGenerationAnswer(strings.TrimSpace(r.Answer), c.Expected) == nil
					for name, content := range evals.CommandGenerationFiles() {
						data, err := os.ReadFile(filepath.Join(root, filepath.FromSlash(name)))
						if err != nil || string(data) != content {
							correct = false
						}
					}
					fileCount := 0
					err = filepath.WalkDir(root, func(path string, entry fs.DirEntry, err error) error {
						if err != nil {
							return err
						}
						if !entry.IsDir() {
							fileCount++
						}
						return nil
					})
					if err != nil || fileCount != len(evals.CommandGenerationFiles()) {
						correct = false
					}
					if len(c.Expected) > 0 {
						r.Outcome.Correct = &correct
					}
					// Quality remains observational. Setup/transport errors and
					// correctness failures stay explicit in every saved report.
				})
			}
		}
	}
	require.NotEmpty(t, reports, "case filter matched no scenarios")
}
