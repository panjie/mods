//go:build integration && windows

package app

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"sync/atomic"
	"testing"

	"github.com/panjie/mods/internal/config"
	"github.com/panjie/mods/internal/evals"
	"github.com/panjie/mods/internal/proto"
	"github.com/panjie/mods/internal/stream"
	toolregistry "github.com/panjie/mods/internal/tools"
	"github.com/stretchr/testify/require"
)

// These commands were observed in Windows P5 sessions with a typed automatic
// control event. Only the production parser/reviewer runs here: every dispatch
// goes to a counter, never to a shell. This is deterministic replay evidence,
// not an instrumented dispatch count from the original model sessions.
func TestCommandGenerationP5WindowsObservedReorganizations(t *testing.T) {
	var fixtures []struct {
		ID      string `json:"id"`
		Case    string `json:"case"`
		Minimal bool   `json:"minimal"`
		Calls   []struct {
			Name string          `json:"name"`
			Args json.RawMessage `json:"args"`
			Want string          `json:"want"`
		} `json:"calls"`
	}
	data, err := os.ReadFile(filepath.Join("testdata", "command_generation", "p5_windows_reorganizations.json"))
	require.NoError(t, err)
	require.NoError(t, json.Unmarshal(data, &fixtures))
	require.NotEmpty(t, fixtures)
	cases, err := evals.CommandGenerationCases()
	require.NoError(t, err)
	for _, fixture := range fixtures {
		t.Run(fixture.ID, func(t *testing.T) {
			cfg := defaultConfig()
			cfg.WorkingDir = t.TempDir()
			cfg.Minimal = fixture.Minimal
			cfg.CommandSimplification = config.CommandSimplificationOnce
			cfg.ReviewMode = ReviewAuto
			cfg.InteractiveTTYAvailable = false
			for name, content := range evals.CommandGenerationFiles() {
				path := filepath.Join(cfg.WorkingDir, filepath.FromSlash(name))
				require.NoError(t, os.MkdirAll(filepath.Dir(path), 0700))
				require.NoError(t, os.WriteFile(path, []byte(content), 0600))
			}
			var prompt string
			for _, c := range cases {
				if c.ID == fixture.Case {
					prompt = c.Prompt
				}
			}
			require.NotEmpty(t, prompt)
			m := &Mods{Config: &cfg, ctx: context.Background(), reviewer: newToolReviewer(&cfg)}
			state := newCommandOrganizationState(m.ctx, prompt)
			defer state.close()
			state.bindSourceContext([]proto.Message{{Role: proto.RoleUser, Content: prompt}})
			var dispatches atomic.Int32
			controls := 0
			for i, call := range fixture.Calls {
				registry := organizationRegistry(t, call.Name, toolregistry.ToolKindShell, &dispatches)
				caller := m.toolResultCallerForOrganization(registry, &cfg, state)
				id := fmt.Sprintf("replay-%d", i)
				before := dispatches.Load()
				msg, status := stream.CallTool(proto.ToolCallRequest{ID: id, Name: call.Name, Arguments: call.Args}, nil, caller)
				require.Len(t, msg.ToolCalls, 1)
				require.Equal(t, id, msg.ToolCalls[0].ID)
				switch call.Want {
				case "reorganize":
					var control *commandReorganizationError
					require.ErrorAs(t, status.Err, &control)
					require.True(t, control.Automatic)
					require.True(t, msg.ToolCalls[0].IsError)
					require.Contains(t, msg.Content, `"status":"not_executed"`)
					require.Equal(t, before, dispatches.Load())
					controls++
				case "ask":
					require.ErrorIs(t, status.Err, errReviewUnavailable)
					require.Equal(t, before, dispatches.Load())
				case "allow":
					require.NoError(t, status.Err)
					require.Equal(t, before+1, dispatches.Load())
				default:
					t.Fatalf("invalid fixture expectation: %s", call.Want)
				}
				t.Logf("call=%d want=%s mock_dispatch_delta=%d", i, call.Want, dispatches.Load()-before)
			}
			require.Equal(t, 1, controls)
		})
	}
}
